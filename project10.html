                        import time
import board
import neopixel

RED = (255, 0, 0)
OFF = (0, 0, 0)

right_eye = neopixel.NeoPixel(board.D8, 4, brightness=0.5, auto_write=False)
left_eye = neopixel.NeoPixel(board.D7, 4, brightness=0.5, auto_write=False)

def show_eyes(colors):
    reversed_colors = colors[::-1]       # left eye gets the list backwards
    for i in range(4):
        right_eye[i] = colors[i]
        left_eye[i] = reversed_colors[i]
    right_eye.show()
    left_eye.show()

OPEN = [RED, RED, RED, RED]     # all 4 lit
HALF = [OFF, OFF, RED, RED]     # bottom row only (squint)
CLOSED = [OFF, OFF, OFF, OFF]   # all off

def blink():
    show_eyes(HALF)
    time.sleep(0.08)
    show_eyes(CLOSED)
    time.sleep(0.12)
    show_eyes(HALF)
    time.sleep(0.08)
    show_eyes(OPEN)

show_eyes(OPEN)

while True:
    time.sleep(2.5)    # eyes stay open
    blink()
    time.sleep(0.4)
    blink()            # quick double blink
